% ── SLIDE 3: ¿Qué NO es un Artículo? ─────────────────────────
% Sugerencia: Incluir ícono de artículo científico o pluma tachada
\begin{frame}{¿Qué NO es un Artículo?}
  \framesubtitle{Definición en negativo: una sola idea y evidencia}
  \vspace{-2pt}

  \begin{itemize}
    \item No es el manual de la herramienta que desarrollaste.
    \item No es una narración de todo lo que hiciste durante dos años.
    \item No es propaganda de tu proyecto.
  \end{itemize}

  \vspace{0.5cm}
  {\color{OffWhite}\rmfamily\bfseries\normalsize Un artículo es la comunicación objetiva de una sola idea y la evidencia que la respalda.}
  \vspace{0.3cm}
  {\color{Cloud}\sffamily\fontsize{8}{10}\selectfont Esa restricción — entre ocho y doce páginas — es lo que lo hace difícil.}
\end{frame}

% ── SLIDE 4: Tres Preguntas Obligatorias ─────────────────────
\begin{frame}{Tres Preguntas Obligatorias}
  \framesubtitle{De Snyder (1991): antes de escribir una línea}
  \vspace{-2pt}

  \begin{enumerate}
    \item ¿Por qué escribo este artículo? (Si es 'para documentar lo que hice', es un informe técnico.)
    \item ¿Qué estoy comunicando? (Debe resumirse en una oración.)
    \item ¿A quién va dirigido? (Define el vehículo, nivel técnico y términos.)
  \end{enumerate}

  \vspace{0.5cm}
  {\color{Ember}\sffamily\bfseries Si no puedes responder con precisión alguna, el artículo aún no está listo.}
\end{frame}

% ── SLIDE 6: Tipos Principales de Artículos ───────────────────
\begin{frame}{Tipos Principales de Artículos}
  \framesubtitle{En Ciencias de la Computación}
  \vspace{-2pt}

  \begin{itemize}
    \item {\bfseries Teórico:} Definiciones + pruebas lógicas formales.
    \item {\bfseries Reporte de Experiencia:} Generalización de observaciones, no narración cronológica.
    \item {\bfseries Sobre Métodos:} Comparación objetiva con métricas claras + reconocimiento de limitaciones.
  \end{itemize}
\end{frame}

% ── SLIDE 12: Plagio: Dos Formas ─────────────────────────────

  \framesubtitle{Un mismo problema}
  \vspace{-2pt}

  \begin{itemize}
    \item {\bfseries Copia Literal:} Texto ajeno sin comillas ni fuente.
    \item {\bfseries Copia de Ideas:} Mismas ideas en secuencia lógica, aunque parafraseadas.
  \end{itemize}

  \vspace{0.5cm}
  {\color{Ember}\sffamily\bfseries Regla básica inamovible: comillas + fuente, siempre, sin excepción.}

% ── SLIDE 13: Marco Legal en Costa Rica ──────────────────────
\begin{frame}{Marco Legal en Costa Rica}
  \framesubtitle{Ley No. 6683: plagio genera responsabilidad civil, no penal}
  \vspace{-2pt}

  \begin{itemize}
    \item Protección automática desde la creación (Reglamento 24611-J, Art. 2).
    \item Duración: vida del autor + 70 años (Art. 58).
    \item Excepción académica (Art. 73): usos honrados con fines educativos, parcialmente y con fuente.
  \end{itemize}
\end{frame}

% ── SLIDE 14: Sanciones en Universidades ─────────────────────
\begin{frame}{Sanciones en Universidades}
  \framesubtitle{Públicas costarricenses (Scielo CR, 2022)}
  \vspace{-2pt}

  \begin{itemize}
    \item {\bfseries UCR:} Falta grave (obra sin protección) / muy grave (obra protegida). Personal académico desde 2014.
    \item {\bfseries TEC:} 'Conducta fraudulenta'; sanciones progresivas.
    \item {\bfseries UNED:} Para posgrado, cualquier plagio es falta muy grave.
  \end{itemize}
\end{frame}

% ── SLIDE 15: Autoplagio y Envíos Simultáneos ────────────────
\begin{frame}{Autoplagio y Envíos Simultáneos}
  \framesubtitle{Dos reglas finales}
  \vspace{-2pt}

  \begin{itemize}
    \item {\bfseries Autoplagio:} Un artículo, un vehículo a la vez. Válido reutilizar texto propio de monografía origen.
    \item {\bfseries Envíos Simultáneos:} No múltiples versiones del mismo artículo.
  \end{itemize}
\end{frame}

% ── SLIDE 16: Lista de Verificación ───────────────────────────
\begin{frame}{Lista de Verificación}
  \framesubtitle{Antes de enviar}
  \vspace{-2pt}

  \begin{itemize}
    \item ¿Puedo resumir la contribución en una oración?
    \item ¿El abstract declara el resultado, no solo el tema?
    \item ¿Comparo con el estado del arte actual?
    \item ¿Menciono trabajos relacionados con comparación objetiva?
    \item ¿Todo texto ajeno está entre comillas y fuente?
    \item ¿El artículo va a un único vehículo?
    \item ¿El vehículo es adecuado al nivel de la contribución?
  \end{itemize}
\end{frame}
